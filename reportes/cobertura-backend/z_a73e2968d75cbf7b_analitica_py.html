58def validar_evento(evento: str, props: dict) -> None: # verifica nombre, propiedades y valores de un evento antes de guardarlo 

59 if evento not in EVENTOS_PERMITIDOS: # si el evento no está definido 

60 raise ValueError(f"Evento desconocido: {evento}") # se rechaza 

61 permitidas = EVENTOS_PERMITIDOS[evento] # propiedades que acepta este evento 

62 sobrantes = set(props) - set(permitidas) # propiedades que no fueron definidas para este evento 

63 if sobrantes: # si llegó algún dato que la pregunta no necesita 

64 raise ValueError(f"Propiedades no permitidas: {', '.join(sorted(sobrantes))}") # se rechaza (privacidad) 

65 for clave, valor in props.items(): # revisa el valor de cada propiedad 

66 valores = permitidas[clave] # valores aceptados para esta propiedad 

67 if valores == BOOLEANO: # si es una propiedad de sí/no 

68 if not isinstance(valor, bool): # y no llegó True o False 

69 raise ValueError(f"{clave} debe ser verdadero o falso") # se rechaza 

70 elif valor not in valores: # si el valor no está en la lista fija 

71 raise ValueError(f"{clave} no está en la lista permitida") # se rechaza (evita texto libre del usuario)

74def porcentaje_terminan(iniciados: int, guardados: int) -> int: # calcula qué % de los que empiezan terminan 

75 if iniciados <= 0: # si nadie empezó todavía 

76 return 0 # evita dividir entre cero 

77 return int(guardados * 100 / iniciados + 0.5) # porcentaje redondeado a entero (.5 sube, como Math.round)

80def caida_entre_pasos(anterior: int, siguiente: int) -> int: # % de personas que se pierden entre un paso y el siguiente 

81 if anterior <= 0: # si nadie llegó al paso anterior 

82 return 0 # no hay caída que medir 

83 return int(max(anterior - siguiente, 0) * 100 / anterior + 0.5) # porcentaje perdido, redondeado

86def armar_embudos(conteos: dict[str, int], errores: dict[tuple[str, str], int]) -> list[dict]: # arma un embudo por flujo 

87 embudos = [] # lista de embudos 

88 for flujo in FLUJOS: # recorre cada flujo principal 

89 pasos = [] # pasos de este embudo 

90 anterior = None # total del paso anterior 

91 for evento in flujo["pasos"]: # recorre los pasos en orden 

92 total = conteos.get(evento, 0) # cuántos llegaron a este paso 

93 caida = caida_entre_pasos(anterior, total) if anterior is not None else 0 # caída desde el paso anterior 

94 pasos.append({"evento": evento, "total": total, "caida": caida}) # agrega el paso 

95 anterior = total # este paso pasa a ser el anterior 

96 inicio = pasos[0]["total"] # cuántos empezaron 

97 exito = pasos[-1]["total"] # cuántos terminaron 

98 motivos = [ # errores del flujo agrupados por motivo 

99 {"motivo": motivo, "total": total} # un motivo y cuántas veces pasó 

100 for (evento, motivo), total in sorted(errores.items()) # recorre todos los errores 

101 if evento == flujo["error"] # solo los de este flujo 

102 ] # fin de motivos 

103 embudos.append({ # agrega el embudo completo 

104 "id": flujo["id"], # identificador del flujo 

105 "nombre": flujo["nombre"], # nombre legible 

106 "pasos": pasos, # pasos con total y caída 

107 "porcentaje_terminan": porcentaje_terminan(inicio, exito), # % que llega al éxito 

108 "evento_error": flujo["error"], # evento de error o salida del flujo 

109 "total_errores": conteos.get(flujo["error"], 0), # cuántas veces se frenó 

110 "motivos": motivos, # errores por motivo 

111 }) # fin del embudo
